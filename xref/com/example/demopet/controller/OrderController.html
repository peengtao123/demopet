1   package com.example.demopet.controller;
2   
3   import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
4   import com.baomidou.mybatisplus.extension.plugins.pagination.Page;
5   import com.example.demopet.entity.Order;
6   import com.example.demopet.service.OrderService;
7   import org.springframework.beans.factory.annotation.Autowired;
8   import org.springframework.stereotype.Controller;
9   import org.springframework.ui.Model;
10  import org.springframework.web.bind.annotation.*;
11  
12  import java.time.LocalDateTime;
13  
14  @Controller
15  @RequestMapping("/order")
16  public class OrderController {
17  
18      @Autowired
19      private OrderService orderService;
20  
21      @GetMapping
22      public String list(@RequestParam(defaultValue = "1") int pageNum,
23                         @RequestParam(defaultValue = "10") int pageSize,
24                         @RequestParam(required = false) String orderNo,
25                         @RequestParam(required = false) Integer status,
26                         Model model) {
27          Page<Order> page = new Page<>(pageNum, pageSize);
28          LambdaQueryWrapper<Order> wrapper = new LambdaQueryWrapper<>();
29          
30          if (orderNo != null && !orderNo.isEmpty()) {
31              wrapper.like(Order::getOrderNo, orderNo);
32          }
33          
34          if (status != null) {
35              wrapper.eq(Order::getStatus, status);
36          }
37          
38          wrapper.orderByDesc(Order::getCreateTime);
39          orderService.page(page, wrapper);
40          
41          model.addAttribute("page", page);
42          model.addAttribute("orderNo", orderNo);
43          model.addAttribute("status", status);
44          return "order/index";
45      }
46  
47      @GetMapping("/detail/{id}")
48      public String detail(@PathVariable Long id, Model model) {
49          Order order = orderService.getById(id);
50          model.addAttribute("order", order);
51          return "order/detail";
52      }
53  
54      @PostMapping("/updateStatus")
55      public String updateStatus(@RequestParam Long id, @RequestParam Integer status) {
56          Order order = orderService.getById(id);
57          if (order != null) {
58              order.setStatus(status);
59              order.setUpdateTime(LocalDateTime.now());
60              orderService.updateById(order);
61          }
62          return "redirect:/order";
63      }
64  
65      @PostMapping("/delete/{id}")
66      public String delete(@PathVariable Long id) {
67          orderService.removeById(id);
68          return "redirect:/order";
69      }
70  }
