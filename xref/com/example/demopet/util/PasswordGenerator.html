1   package com.example.demopet.util;

3   import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
4   
5   /**
6    * BCrypt 密码生成工具
7    * 运行此类的 main 方法可以生成密码的 BCrypt 哈希值
8    */
9   public class PasswordGenerator {
10      
11      public static void main(String[] args) {
12          BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();
13          
14          // 要加密的原始密码
15          String rawPassword = "123456";
16          
17          // 生成 BCrypt 哈希（每次运行结果都不同，但都能验证通过）
18          String encodedPassword = encoder.encode(rawPassword);
19          
20          System.out.println("========================================");
21          System.out.println("原始密码: " + rawPassword);
22          System.out.println("BCrypt 哈希: " + encodedPassword);
23          System.out.println("========================================");
24          System.out.println();
25          
26          // 验证生成的哈希是否正确
27          boolean matches = encoder.matches(rawPassword, encodedPassword);
28          System.out.println("密码验证结果: " + (matches ? "✓ 成功" : "✗ 失败"));
29          System.out.println();
30          
31          // 生成 SQL 更新语句
32          System.out.println("SQL 更新语句：");
33          System.out.println("UPDATE user SET password = '" + encodedPassword + "' WHERE username = 'admin';");
34          System.out.println();
35          
36          // 生成完整的 INSERT 语句
37          System.out.println("完整的数据初始化 SQL：");
38          System.out.println("INSERT INTO `user` (id, username, password, name, email, phone, status) VALUES");
39          System.out.println("(1, 'admin', '" + encodedPassword + "', '超级管理员', 'admin@example.com', '13800138000', 1),");
40          System.out.println("(2, 'manager', '" + encodedPassword + "', '管理员', 'manager@example.com', '13800138001', 1),");
41          System.out.println("(3, 'user1', '" + encodedPassword + "', '张三', 'zhangsan@example.com', '13800138002', 1),");
42          System.out.println("(4, 'user2', '" + encodedPassword + "', '李四', 'lisi@example.com', '13800138003', 1),");
43          System.out.println("(5, 'user3', '" + encodedPassword + "', '王五', 'wangwu@example.com', '13800138004', 1);");
44      }
45  }
