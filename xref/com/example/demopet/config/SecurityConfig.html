1   package com.example.demopet.config;
2   
3   import com.example.demopet.service.CustomUserDetailsService;
4   import org.springframework.beans.factory.annotation.Autowired;
5   import org.springframework.context.annotation.Bean;
6   import org.springframework.context.annotation.Configuration;
7   import org.springframework.security.authentication.AuthenticationManager;
8   import org.springframework.security.authentication.dao.DaoAuthenticationProvider;
9   import org.springframework.security.config.annotation.authentication.configuration.AuthenticationConfiguration;
10  import org.springframework.security.config.annotation.web.builders.HttpSecurity;
11  import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
12  import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
13  import org.springframework.security.crypto.password.PasswordEncoder;
14  import org.springframework.security.web.SecurityFilterChain;
15  
16  @Configuration
17  @EnableWebSecurity
18  public class SecurityConfig {
19      
20      @Autowired
21      private CustomUserDetailsService userDetailsService;
22      
23      /**
24       * 密码编码器 - 使用 BCrypt
25       */
26      @Bean
27      public PasswordEncoder passwordEncoder() {
28          return new BCryptPasswordEncoder();
29      }
30      
31      /**
32       * 认证提供者
33       */
34      @Bean
35      public DaoAuthenticationProvider authenticationProvider() {
36          DaoAuthenticationProvider authProvider = new DaoAuthenticationProvider(userDetailsService);
37          authProvider.setPasswordEncoder(passwordEncoder());
38          return authProvider;
39      }
40      
41      /**
42       * 认证管理器
43       */
44      @Bean
45      public AuthenticationManager authenticationManager(AuthenticationConfiguration authConfig) throws Exception {
46          return authConfig.getAuthenticationManager();
47      }
48      
49      /**
50       * 安全过滤链配置
51       */
52      @Bean
53      public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
54          http
55              // 禁用 CSRF（如果需要可以启用）
56              .csrf(csrf -> csrf.disable())
57              
58              // 授权配置
59              .authorizeHttpRequests(auth -> auth
60                  // 公开访问的路径
61                  .requestMatchers("/login", "/", "/error", "/css/**", "/js/**", "/images/**").permitAll()
62                  // 需要认证的路径
63                  .requestMatchers("/dashboard", "/user/**", "/role/**", "/menu/**", "/product/**", "/order/**").authenticated()
64                  // 其他路径都需要认证
65                  .anyRequest().authenticated()
66              )
67              
68              // 表单登录配置
69              .formLogin(form -> form
70                  .loginPage("/login")
71                  .loginProcessingUrl("/login")
72                  .defaultSuccessUrl("/dashboard", true)
73                  .failureUrl("/login?error=true")
74                  .usernameParameter("username")
75                  .passwordParameter("password")
76                  .permitAll()
77              )
78              
79              // 登出配置
80              .logout(logout -> logout
81                  .logoutUrl("/logout")
82                  .logoutSuccessUrl("/login?logout=true")
83                  .invalidateHttpSession(true)
84                  .deleteCookies("JSESSIONID")
85                  .permitAll()
86              )
87              
88              // 会话管理
89              .sessionManagement(session -> session
90                  .maximumSessions(1)
91                  .maxSessionsPreventsLogin(false)
92              );
93          
94          return http.build();
95      }
96  }
