3   import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
4   import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
5   import com.example.demopet.entity.UserRole;
6   import com.example.demopet.mapper.UserRoleMapper;
7   import org.springframework.stereotype.Service;
8   import org.springframework.transaction.annotation.Transactional;
9   
10  import java.util.List;
11  import java.util.stream.Collectors;
12  
13  @Service
14  public class UserRoleService extends ServiceImpl<UserRoleMapper, UserRole> {
15      
16      /**
17       * 根据用户ID获取角色ID列表
18       */
19      @Transactional(readOnly = true)
20      public List<Long> getRoleIdsByUserId(Long userId) {
21          LambdaQueryWrapper<UserRole> wrapper = new LambdaQueryWrapper<>();
22          wrapper.eq(UserRole::getUserId, userId);
23          List<UserRole> userRoles = this.list(wrapper);
24          return userRoles.stream()
25                  .map(UserRole::getRoleId)
26                  .collect(Collectors.toList());
27      }
28      
29      /**
30       * 为用户分配角色（先删除旧关系，再添加新关系）
31       */
32      @Transactional
33      public void assignRolesToUser(Long userId, List<Long> roleIds) {
34          // 删除用户原有的所有角色
35          LambdaQueryWrapper<UserRole> deleteWrapper = new LambdaQueryWrapper<>();
36          deleteWrapper.eq(UserRole::getUserId, userId);
37          this.remove(deleteWrapper);
38          
39          // 添加新的角色关联
40          if (roleIds != null && !roleIds.isEmpty()) {
41              for (Long roleId : roleIds) {
42                  UserRole userRole = new UserRole();
43                  userRole.setUserId(userId);
44                  userRole.setRoleId(roleId);
45                  this.save(userRole);
46              }
47          }
48      }
49  }
