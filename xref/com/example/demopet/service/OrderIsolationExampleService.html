19  /**
20   * 订单服务 - 演示不同隔离级别的应用
21   */
22  @Service
23  public class OrderIsolationExampleService {
24      
25      @Autowired
26      private OrderMapper orderMapper;
27      
28      @Autowired
29      private OrderItemMapper orderItemMapper;
30      
31      @Autowired
32      private ProductMapper productMapper;
33      
34      /**
35       * 场景1：创建订单 - 使用默认隔离级别
36       * 大多数业务场景使用 DEFAULT 即可（MySQL 下是 REPEATABLE_READ）
37       */
38      @Transactional(isolation = Isolation.DEFAULT, rollbackFor = Exception.class)
39      public Order createOrder(Order order, List<OrderItem> items) {
40          // 1. 保存订单
41          order.setCreateTime(LocalDateTime.now());
42          orderMapper.insert(order);
43          
44          // 2. 保存订单项
45          for (OrderItem item : items) {
46              item.setOrderId(order.getId());
47              orderItemMapper.insert(item);
48          }
49          
50          return order;
51      }
52      
53      /**
54       * 场景2：查询订单列表 - 使用读已提交
55       * 报表查询不需要严格一致性，READ_COMMITTED 性能更好
56       */
57      @Transactional(isolation = Isolation.READ_COMMITTED, readOnly = true)
58      public List<Order> queryOrdersWithReadCommitted() {
59          // 只读查询，使用较低的隔离级别提升并发性能
60          LambdaQueryWrapper<Order> wrapper = new LambdaQueryWrapper<>();
61          wrapper.orderByDesc(Order::getCreateTime);
62          return orderMapper.selectList(wrapper);
63      }
64      
65      /**
66       * 场景3：统计订单数据 - 使用读已提交 + 只读
67       * 统计数据允许轻微的不一致，追求查询性能
68       */
69      @Transactional(isolation = Isolation.READ_COMMITTED, readOnly = true)
70      public OrderStatistics getOrderStatistics() {
71          // 统计总订单数
72          long totalOrders = orderMapper.selectCount(null);
73          
74          // 统计总金额
75          List<Order> orders = orderMapper.selectList(null);
76          BigDecimal totalAmount = orders.stream()
77                  .map(Order::getTotalAmount)
78                  .reduce(BigDecimal.ZERO, BigDecimal::add);
79          
80          return new OrderStatistics(totalOrders, totalAmount);
81      }
82      
83      /**
84       * 场景4：扣减库存并创建订单 - 使用可重复读
85       * 库存操作需要保证一致性，避免超卖
86       */
87      @Transactional(isolation = Isolation.REPEATABLE_READ, rollbackFor = Exception.class)
88      public Order createOrderWithStockDeduct(Order order, List<OrderItem> items) {
89          // 1. 检查并扣减库存
90          for (OrderItem item : items) {
91              Product product = productMapper.selectById(item.getProductId());
92              
93              if (product == null) {
94                  throw new RuntimeException("商品不存在：" + item.getProductId());
95              }
96              
97              if (product.getStock() < item.getQuantity()) {
98                  throw new RuntimeException("库存不足：" + product.getProductName());
99              }
100             
101             // 扣减库存
102             product.setStock(product.getStock() - item.getQuantity());
103             product.setSales(product.getSales() + item.getQuantity());
104             productMapper.updateById(product);
105         }
106         
107         // 2. 创建订单
108         order.setCreateTime(LocalDateTime.now());
109         orderMapper.insert(order);
110         
111         // 3. 创建订单项
112         for (OrderItem item : items) {
113             item.setOrderId(order.getId());
114             orderItemMapper.insert(item);
115         }
116         
117         return order;
118     }

120     /**
121      * 场景5：高并发库存扣减 - 使用串行化（谨慎使用）
122      * 极端情况下使用，完全避免并发问题
123      * 注意：会导致严重的性能问题，仅用于关键业务
124      */
125     @Transactional(isolation = Isolation.SERIALIZABLE, rollbackFor = Exception.class)
126     public void deductStockSerializably(Long productId, int quantity) {
127         Product product = productMapper.selectById(productId);
128         
129         if (product == null) {
130             throw new RuntimeException("商品不存在");
131         }
132         
133         if (product.getStock() < quantity) {
134             throw new RuntimeException("库存不足");
135         }
136         
137         // 扣减库存
138         product.setStock(product.getStock() - quantity);
139         productMapper.updateById(product);
140         
141         // 在 SERIALIZABLE 级别下，其他事务必须等待这个事务完成
142         // 这确保了不会出现超卖，但会严重影响并发性能
143     }

145     /**
146      * 场景6：批量更新订单状态 - 使用默认隔离级别
147      */
148     @Transactional(rollbackFor = Exception.class)
149     public void batchUpdateOrderStatus(List<Long> orderIds, Integer status) {
150         for (Long orderId : orderIds) {
151             Order order = orderMapper.selectById(orderId);
152             if (order != null) {
153                 order.setStatus(status);
154                 order.setUpdateTime(LocalDateTime.now());
155                 orderMapper.updateById(order);
156             }
157         }
158     }

160     /**
161      * 场景7：取消订单并恢复库存 - 使用可重复读
162      */
163     @Transactional(isolation = Isolation.REPEATABLE_READ, rollbackFor = Exception.class)
164     public void cancelOrderAndRestoreStock(Long orderId) {
165         // 1. 查询订单
166         Order order = orderMapper.selectById(orderId);
167         if (order == null) {
168             throw new RuntimeException("订单不存在");
169         }
170         
171         if (order.getStatus() != 0 && order.getStatus() != 1) {
172             throw new RuntimeException("订单状态不允许取消");
173         }
174         
175         // 2. 更新订单状态
176         order.setStatus(4); // 已取消
177         order.setUpdateTime(LocalDateTime.now());
178         orderMapper.updateById(order);
179         
180         // 3. 恢复库存
181         List<OrderItem> items = orderItemMapper.selectByOrderId(orderId);
182         for (OrderItem item : items) {
183             Product product = productMapper.selectById(item.getProductId());
184             if (product != null) {
185                 product.setStock(product.getStock() + item.getQuantity());
186                 product.setSales(product.getSales() - item.getQuantity());
187                 productMapper.updateById(product);
188             }
189         }
190     }
