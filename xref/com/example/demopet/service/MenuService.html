10  import java.util.*;
11  import java.util.stream.Collectors;
12  
13  @Service
14  public class MenuService extends ServiceImpl<MenuMapper, Menu> {
15      
16      @Transactional(readOnly = true)
17      public List<Menu> listWithSort() {
18          LambdaQueryWrapper<Menu> wrapper = new LambdaQueryWrapper<>();
19          wrapper.orderByAsc(Menu::getSort);
20          return this.list(wrapper);
21      }
22      
23      @Transactional(readOnly = true)
24      public List<Menu> search(String keyword) {
25          LambdaQueryWrapper<Menu> wrapper = new LambdaQueryWrapper<>();
26          if (keyword != null && !keyword.trim().isEmpty()) {
27              wrapper.like(Menu::getMenuName, keyword)
28                     .or()
29                     .like(Menu::getMenuCode, keyword);
30          }
31          wrapper.orderByAsc(Menu::getSort);
32          return this.list(wrapper);
33      }
34      
35      /**
36       * 获取树形菜单结构
37       */
38      @Transactional(readOnly = true)
39      public List<Menu> getMenuTree() {
40          List<Menu> allMenus = listWithSort();
41          return buildMenuTree(allMenus, 0L);
42      }
43      
44      /**
45       * 构建树形菜单结构
46       */
47      private List<Menu> buildMenuTree(List<Menu> allMenus, Long parentId) {
48          List<Menu> treeMenus = new ArrayList<>();
49          for (Menu menu : allMenus) {
50              if (parentId.equals(menu.getParentId())) {
51                  // 递归查找子菜单
52                  List<Menu> children = buildMenuTree(allMenus, menu.getId());
53                  // 可以通过添加一个临时字段来存储子菜单，但为了简单起见，我们只在需要时查询
54                  treeMenus.add(menu);
55              }
56          }
57          return treeMenus;
58      }
59      
60      /**
61       * 获取父菜单列表（用于表单选择）
62       */
63      @Transactional(readOnly = true)
64      public List<Menu> getParentMenus() {
65          LambdaQueryWrapper<Menu> wrapper = new LambdaQueryWrapper<>();
66          wrapper.eq(Menu::getParentId, 0)
67                 .or()
68                 .isNull(Menu::getParentId)
69                 .orderByAsc(Menu::getSort);
70          return this.list(wrapper);
71      }
72      
73      /**
74       * 根据父ID获取子菜单
75       */
76      @Transactional(readOnly = true)
77      public List<Menu> getChildMenus(Long parentId) {
78          LambdaQueryWrapper<Menu> wrapper = new LambdaQueryWrapper<>();
79          wrapper.eq(Menu::getParentId, parentId)
80                 .orderByAsc(Menu::getSort);
81          return this.list(wrapper);
82      }
83  }
