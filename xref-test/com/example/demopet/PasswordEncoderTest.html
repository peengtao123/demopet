11          
12          // 生成密码 "123456" 的 BCrypt 哈希

15          
16          System.out.println("原始密码: " + rawPassword);
17          System.out.println("BCrypt 哈希: " + encodedPassword);
18          System.out.println();
19          
20          // 验证密码
21          boolean matches = encoder.matches(rawPassword, encodedPassword);
22          System.out.println("密码验证结果: " + matches);
23          
24          // 验证旧的哈希值

27          System.out.println("旧哈希值验证结果: " + oldMatches);
28      }
